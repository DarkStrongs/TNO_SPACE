# 经济周结算：公众支持度向 50 回落，按公众支持度累计拨款
NSP_econ_weekly = {
	set_temp_variable = { nsp_d = 50 }
	subtract_from_temp_variable = { nsp_d = NSP_popularity }
	divide_temp_variable = { nsp_d = 100 }
	add_to_variable = { NSP_popularity = nsp_d }
	NSP_budget_weekly = yes   # 拨款改为每 26 周发一次
}

# 每周：按当周公众支持度累计额度，满 26 周（约半年）一次拨付
NSP_budget_weekly = {
	if = { limit = { NOT = { has_variable = NSP_pay_weeks } } set_variable = { NSP_pay_weeks = 0 } set_variable = { NSP_pay_accum = 0 } set_variable = { NSP_pay_last = 0 } }
	# 每周额度（百万美元）= 30 + 公众支持度 × 0.8 → 支持 50% 时约 $70M/周，半年约 $1.8B
	set_variable = { NSP_income = NSP_popularity }
	multiply_variable = { NSP_income = 0.8 }
	add_to_variable = { NSP_income = 30 }
	round_variable = NSP_income
	NSP_dir_income = yes   # 局长特质：华盛顿老手
	add_to_variable = { NSP_pay_accum = NSP_income }
	add_to_variable = { NSP_pay_weeks = 1 }
	if = {
		limit = { check_variable = { NSP_pay_weeks > 25 } }
		set_variable = { NSP_pay_last = NSP_pay_accum }
		add_to_variable = { NSP_space_budget = NSP_pay_accum }
		set_variable = { NSP_pay_accum = 0 }
		set_variable = { NSP_pay_weeks = 0 }
		NSP_budget_disp = yes
		country_event = { id = nsp.12 }
	}
	NSP_budget_disp = yes
}

	set_variable = { NSP_left_b = NSP_calc_left }         divide_variable = { NSP_left_b = 1000 }
	# 下次拨款预估 = 已累计 + 本周额度 × 剩余周数
	# 开局第一周结算前 NSP_income 还没算，预估会显示 0 → 这里按当前公众支持度现算
	set_variable = { NSP_income = NSP_popularity }

	add_to_variable = { NSP_income = 30 }

# ln(x)：HOI4 没有对数函数。输入：变量 NSP_arg_ln_in（> 0）；输出：变量 NSP_arg_ln_out
#   1) 区间缩减：x > 1.5 时反复 /2，并累加 ln2 = 0.693
#   2) 级数：ln(x) = 2*(y + y^3/3 + y^5/5 + y^7/7)，y = (x-1)/(x+1)，|y| < 0.2 时误差 < 0.001
NSP_ln = {
	set_temp_variable = { ln_x = NSP_arg_ln_in }
	set_temp_variable = { ln_acc = 0 }
	while_loop_effect = {
		limit = { check_variable = { ln_x > 1.5 } }
		divide_temp_variable = { ln_x = 2 }
		add_to_temp_variable = { ln_acc = 0.693 }
	}
	set_temp_variable = { ln_y = ln_x }
	subtract_from_temp_variable = { ln_y = 1 }
	set_temp_variable = { ln_den = ln_x }
	add_to_temp_variable = { ln_den = 1 }
	divide_temp_variable = { ln_y = ln_den }
	set_temp_variable = { ln_y2 = ln_y }
	multiply_temp_variable = { ln_y2 = ln_y }
	set_variable = { NSP_arg_ln_out = ln_y }
	set_temp_variable = { ln_term = ln_y }
	multiply_temp_variable = { ln_term = ln_y2 }
	set_temp_variable = { ln_add = ln_term } divide_temp_variable = { ln_add = 3 } add_to_variable = { NSP_arg_ln_out = ln_add }
	multiply_temp_variable = { ln_term = ln_y2 }
	set_temp_variable = { ln_add = ln_term } divide_temp_variable = { ln_add = 5 } add_to_variable = { NSP_arg_ln_out = ln_add }
	multiply_temp_variable = { ln_term = ln_y2 }
	set_temp_variable = { ln_add = ln_term } divide_temp_variable = { ln_add = 7 } add_to_variable = { NSP_arg_ln_out = ln_add }
	multiply_variable = { NSP_arg_ln_out = 2 }
	add_to_variable = { NSP_arg_ln_out = ln_acc }
}
