#  说明文档：documentation/SUBMOD_API.md；示例子模组：documentation/examples/NSP_addon_example/
#
#  ⭐ 所有 NSP_reg_* 都是「按 id 写入」，不是「追加」：
#     - 同一个 id 反复注册只会覆盖成同样的值，可以放进每次刷新都跑的钩子里；
#     - 数组不够长时，中间缺的格子会用「锁定 / 空」的值补齐，所以 id 可以跳着用；
#     - 进度、状态类数组（研发进度、是否已解锁等）只补长度，不会被覆盖。
#  ⭐ 参数一律用临时变量传（set_temp_variable），调用后临时变量自动失效。
#     没写的参数读作 0；需要 -1 之类默认值的参数，文档里都写了。
#  ⭐ id 段约定（防止两个子模组撞号，见 SUBMOD_API.md §2）：
#     载荷 0..49 / 火箭 0..49 / 目的地 0..31 / 研发 0..199 归本体；子模组从 50 / 50 / 16 / 200 起，
#     并在文档的登记表里写上自己占用的号段。

# ---------------------------------------------------------------------------
# 载荷
#   必填：reg_id  reg_mass（吨）  reg_cost（原型成本，百万）  reg_rel（基础可靠性 0..100）
#   选填：reg_icon      任务卡图标 = GFX_NSP_mission_icons 帧号 1..12（帧 = 本体载荷 id + 1，见 NSP_space.gfx 注释；默认 11 科研卫星）
#         reg_crewed    1 = 载人（吃载人加成、有乘组 / 返程阶段、去带 crew_rd 的目的地要研发）
#         reg_slots     载荷类修正槽位 1..4；0 = 默认 2 个，-1 = 没有槽位
#         （必填）reg_unlock_rd 解锁它的研发 id；-1 = 开局就有。⚠ 不写会读成 0 = 研发 0「红石」
#         reg_custom_art 1 = 规划器大图用你自己的 sprite：本地化键 NSP_PLART_<id> 写 sprite 名，
#                       reg_art_frame 写帧号（默认 1）；0 = 用本体图集 GFX_NSP_payload_art 的第 id+1 帧
#   目的地另外用 NSP_reg_allow 登记；名称、徽章写在本地化（NSP_PL_NAME_<id> 等，见文档）。
# ---------------------------------------------------------------------------
NSP_reg_payload = {
	set_temp_variable = { t_rn = reg_id } add_to_temp_variable = { t_rn = 1 }
	while_loop_effect = { limit = { check_variable = { NSP_pl_mass^num < t_rn } } add_to_array = { NSP_pl_mass = 1 } }
	while_loop_effect = { limit = { check_variable = { NSP_pl_cost^num < t_rn } } add_to_array = { NSP_pl_cost = 1 } }
	while_loop_effect = { limit = { check_variable = { NSP_pl_rel^num < t_rn } } add_to_array = { NSP_pl_rel = 0 } }
	while_loop_effect = { limit = { check_variable = { NSP_pl_icon^num < t_rn } } add_to_array = { NSP_pl_icon = 11 } }
	while_loop_effect = { limit = { check_variable = { NSP_pl_next^num < t_rn } } add_to_array = { NSP_pl_next = 1 } }   # 任务编号：只补不改
	while_loop_effect = { limit = { check_variable = { NSP_pl_unl^num < t_rn } } add_to_array = { NSP_pl_unl = 0 } }     # 解锁状态：只补不改
	while_loop_effect = { limit = { check_variable = { NSP_pl_crewed^num < t_rn } } add_to_array = { NSP_pl_crewed = 0 } }
	while_loop_effect = { limit = { check_variable = { NSP_pl_urd^num < t_rn } } add_to_array = { NSP_pl_urd = -2 } }     # -2 = 空号或本体旧代码管
	while_loop_effect = { limit = { check_variable = { NSP_pl_cart^num < t_rn } } add_to_array = { NSP_pl_cart = 0 } }
	while_loop_effect = { limit = { check_variable = { NSP_pl_afr^num < t_rn } } add_to_array = { NSP_pl_afr = 1 } }
	while_loop_effect = { limit = { check_variable = { NSP_pl_pslots^num < t_rn } } add_to_array = { NSP_pl_pslots = 2 } }
	set_variable = { NSP_pl_mass^reg_id = reg_mass }
	set_variable = { NSP_pl_cost^reg_id = reg_cost }
	set_variable = { NSP_pl_rel^reg_id = reg_rel }
	if = { limit = { check_variable = { reg_icon = 0 } } set_temp_variable = { reg_icon = 11 } }
	set_variable = { NSP_pl_icon^reg_id = reg_icon }
	set_variable = { NSP_pl_crewed^reg_id = reg_crewed }
	set_variable = { NSP_pl_urd^reg_id = reg_unlock_rd }
	set_variable = { NSP_pl_cart^reg_id = reg_custom_art }
	if = { limit = { check_variable = { reg_art_frame = 0 } } set_temp_variable = { reg_art_frame = 1 } }
	set_variable = { NSP_pl_afr^reg_id = reg_art_frame }
	if = { limit = { check_variable = { reg_slots = 0 } } set_temp_variable = { reg_slots = 2 } }   # 0 = 默认 2 个，-1 = 没有槽位
	if = { limit = { check_variable = { reg_slots < 0 } } set_temp_variable = { reg_slots = 0 } }
	set_variable = { NSP_pl_pslots^reg_id = reg_slots }
	NSP_api_pad_matrix = yes
	NSP_reg_clear = yes
}

# ---------------------------------------------------------------------------
# 载荷 × 目的地 许可表 NSP_pd_ok^(载荷 × 32 + 目的地)，所以目的地 id 最多 0..31
#   NSP_reg_allow：reg_id = 载荷，reg_dest = 目的地 → 允许
#   NSP_reg_deny ：同上 → 禁止（可以用来改本体规则）
# ---------------------------------------------------------------------------
NSP_api_pad_matrix = {
	set_temp_variable = { t_mn = NSP_pl_mass^num } multiply_temp_variable = { t_mn = 32 }
	while_loop_effect = { limit = { check_variable = { NSP_pd_ok^num < t_mn } } add_to_array = { NSP_pd_ok = 0 } }
}
NSP_reg_allow = {
	NSP_api_pad_matrix = yes
	set_temp_variable = { t_mi = reg_id } multiply_temp_variable = { t_mi = 32 } add_to_temp_variable = { t_mi = reg_dest }
	if = { limit = { check_variable = { NSP_pd_ok^num > t_mi } } set_variable = { NSP_pd_ok^t_mi = 1 } }
}
NSP_reg_deny = {
	NSP_api_pad_matrix = yes
	set_temp_variable = { t_mi = reg_id } multiply_temp_variable = { t_mi = 32 } add_to_temp_variable = { t_mi = reg_dest }
	if = { limit = { check_variable = { NSP_pd_ok^num > t_mi } } set_variable = { NSP_pd_ok^t_mi = 0 } }
}

# 目的地（id 0..31）
#   必填：reg_id  reg_dv（所需 Delta-V）

#         reg_body      卫星显示在哪个天体：3 地球 / 9 月球 / 4 火星 / 2 金星 …（不写按 3 地球）
#                       -1 = 逃逸地球规则（水手号奇数金星、偶数火星，其他火星）；-2 = 不显示卫星
#         reg_view      Mission Control 哪个视图：0 地球 / 1 月球 / 2 行星际
#         reg_row       该视图的第几条轨道行：0 / 1 / 2（界面只有 3 行）
#   必填（-1 = 无）：reg_crew_rd   载人前往需要的研发 id；-1 = 不需要
#                   reg_unlock_rd 解锁它的研发 id；-1 = 开局就有
#         reg_custom_art / reg_art_frame  规划器轨迹图，本地化键 NSP_TRAJART_<id>

	while_loop_effect = { limit = { check_variable = { NSP_dest_crd^num < t_rn } } add_to_array = { NSP_dest_crd = -1 } }
	while_loop_effect = { limit = { check_variable = { NSP_dest_cart^num < t_rn } } add_to_array = { NSP_dest_cart = 0 } }
	while_loop_effect = { limit = { check_variable = { NSP_dest_afr^num < t_rn } } add_to_array = { NSP_dest_afr = 1 } }
	set_variable = { NSP_dest_dv^reg_id = reg_dv }

	set_variable = { NSP_dest_xp^reg_id = reg_xp }

# 把目的地 reg_id 挪到规划器列表里 reg_dest（目的地 id）的后面；reg_dest = -1 → 挪到最前
NSP_reg_dest_order = {
	if = { limit = { is_in_array = { array = NSP_dest_ids value = reg_id } } remove_from_array = { array = NSP_dest_ids value = reg_id } }
	set_temp_variable = { t_ri = 0 }
	if = {
		limit = { NOT = { check_variable = { reg_dest = -1 } } }
		set_temp_variable = { t_ri = NSP_dest_ids^num }
		for_each_loop = { array = NSP_dest_ids value = t_rv index = t_rk  if = { limit = { check_variable = { t_rv = reg_dest } } set_temp_variable = { t_ri = t_rk } add_to_temp_variable = { t_ri = 1 } } }
	}
	add_to_array = { array = NSP_dest_ids value = reg_id index = t_ri }
}

# ---------------------------------------------------------------------------
# 研发项目
#   必填：reg_id（≥ 200 给子模组）  reg_track（分类 0..5）  reg_col（0..7）  reg_row（0..4）
#         reg_cost  reg_days  reg_tech（完成后加的太空科技）
#         reg_req1 / reg_req2  前置研发 id，-1 = 无（必须明写，不写会读成 0 = 研发 0「红石」）
#   卡面：改在本地化写 NSP_RDART_<id>: "GFX_你的图"（239×117 单张；可直接填本体的 GFX_NSP_rdart_<id> 借图）
#   选填：reg_art     旧字段，只存数据
#         reg_bonus   1 = 完成后研发槽 +1
#         reg_uk1 / reg_uv1 / reg_uk2 / reg_uv2  完成时一次性效果（种类 / 数值，种类表见文档；11 = 支持度）
#   格位：研发树格子 分类 × 40 + 行 × 8 + 列 必须是空的（NSP_rd_blank），占了就不放上树，只登记数据。
#   卡片名称 / 描述 / 解锁说明写在本地化 NSP_RD_NAME_<id> / NSP_RD_DESC_<id> / NSP_RD_UNLOCK_<id>。

# 通用解锁：按 *_urd 表，把 urd ≥ -1 的载荷 / 火箭 / 目的地设成解锁或锁定
#   -2 = 本体旧代码自己管（NSP_unlock_payloads_vehicles 等），这里不碰
#   本体在 NSP_rd_apply_unlocks 末尾调用，子模组不用自己调
# ---------------------------------------------------------------------------
NSP_api_apply_unlocks = {
	for_each_loop = {
		array = NSP_pl_urd value = t_ru index = t_ri
		if = {
			limit = { check_variable = { t_ru > -2 } }
			set_variable = { NSP_pl_unl^t_ri = 0 }
			if = { limit = { check_variable = { t_ru = -1 } } set_variable = { NSP_pl_unl^t_ri = 1 } }
			else_if = { limit = { check_variable = { NSP_rd_state^t_ru = 3 } } set_variable = { NSP_pl_unl^t_ri = 1 } }
		}
	}
	for_each_loop = {
		array = NSP_ve_urd value = t_ru index = t_ri
		if = {
			limit = { check_variable = { t_ru > -2 } }
			set_variable = { NSP_ve_unl^t_ri = 0 }
			if = { limit = { check_variable = { t_ru = -1 } } set_variable = { NSP_ve_unl^t_ri = 1 } }
			else_if = { limit = { check_variable = { NSP_rd_state^t_ru = 3 } } set_variable = { NSP_ve_unl^t_ri = 1 } }
		}
	}
	for_each_loop = {
		array = NSP_dest_urd value = t_ru index = t_ri
		if = {
			limit = { check_variable = { t_ru > -2 } }
			set_variable = { NSP_dest_unl^t_ri = 0 }
			if = { limit = { check_variable = { t_ru = -1 } } set_variable = { NSP_dest_unl^t_ri = 1 } }
			else_if = { limit = { check_variable = { NSP_rd_state^t_ru = 3 } } set_variable = { NSP_dest_unl^t_ri = 1 } }
		}
	}
}

#############################################################################
#  事件接口 NSP_ev_*：在事件选项（或决议、国策）里直接调用，不用改本体任何文件
#  说明文档：documentation/EVENT_API.md；示例：documentation/examples/NSP_event_api_example/
#
#  用法：先用临时变量写参数，再调用效果。每个效果自带选项提示（custom_effect_tooltip），
#        实际改动都在 hidden_effect 里，所以选项里不用再写提示。
#    option = {
#        name = my_event.1.a
#        set_temp_variable = { ev_val = 0.5 }  NSP_ev_add_budget = yes      # 经费 +0.5B
#        set_temp_variable = { ev_id = 53 }    NSP_ev_complete_rd = yes     # 立即完成研发 53
#        NSP_ev_this_mission = yes                                           # 选中「本次任务」
#        set_temp_variable = { ev_val = 10 }   NSP_ev_mission_rel = yes     # 本次任务成功率 +10
#    }
#  参数：ev_val 数值（单位见各效果）/ ev_id 研发 id / ev_mid 任务 id（-1 = 没有任务，效果什么也不做）
#        ev_silent = 1 时不显示自带提示（想自己写提示时用）
#  每个效果执行完会把 ev_val、ev_id 清零；ev_mid 保留，选一次任务可以连用几个任务效果。
#############################################################################

# ---- 内部：效果结束后的公共收尾（刷新显示、清参数）----
NSP_ev_finish = {
	hidden_effect = {
		if = { limit = { check_variable = { NSP_popularity > 100 } } set_variable = { NSP_popularity = 100 } }
		if = { limit = { check_variable = { NSP_popularity < 0 } } set_variable = { NSP_popularity = 0 } }
		NSP_budget_disp = yes
		NSP_recalc_home = yes
		if = { limit = { has_country_flag = NSP_planner_open } NSP_recalc_planner = yes }
		add_to_variable = { NSP_dirty = 1 }
	}
	set_temp_variable = { ev_val = 0 }
	set_temp_variable = { ev_id = 0 }
}

		else = { custom_effect_tooltip = NSP_EV_TT_BUDGET_P }

		add_to_variable = { NSP_space_budget = nsp_evm }
	}

}

# 立即完成研发 ev_id（不管前置是否完成、是否在研；已完成的什么也不做）
#   和正常完成一样：给太空科技经验、一次性奖励、研发槽奖励，重算解锁
NSP_ev_complete_rd = {
	if = { limit = { NOT = { check_variable = { ev_silent = 1 } } check_variable = { ev_id > -1 } check_variable = { NSP_rd_state^num > ev_id } NOT = { check_variable = { NSP_rd_state^ev_id = 3 } } } custom_effect_tooltip = NSP_EV_TT_RD_DONE }
	hidden_effect = {
		set_temp_variable = { p = ev_id }
		if = {
			limit = {
				check_variable = { p > -1 }
				check_variable = { NSP_rd_state^num > p }
				NOT = { check_variable = { NSP_rd_state^p = 3 } }
			}
			if = { limit = { is_in_array = { array = NSP_rd_queue value = p } } remove_from_array = { array = NSP_rd_queue value = p } }
			if = { limit = { check_variable = { NSP_rd_bonus^p = 1 } } add_to_variable = { NSP_rd_slots = 1 } }
			set_variable = { NSP_rd_state^p = 3 }
			set_variable = { NSP_rd_prog^p = 100 }
			add_to_variable = { NSP_tech = NSP_rd_tech^p }
			set_variable = { NSP_arg_rd = p }
			NSP_rd_on_complete = yes
			NSP_rd_apply_unlocks = yes
			NSP_tech_tier_check = yes
			NSP_rd_recalc = yes
		}
	}
	NSP_ev_finish = yes
}

# 正在研发的 ev_id 进度 +ev_val（百分点）；到 100 立即完成。没在研发时什么也不做
NSP_ev_rd_progress = {

# 重算 NSP_newest_mid：任务开始、每日结算删除已结束任务、开新局时调用
NSP_newest_mid_update = {
	set_variable = { NSP_newest_mid = -1 }
	for_each_loop = {
		array = NSP_missions value = nsp_evq
		if = { limit = { check_variable = { nsp_evq > NSP_newest_mid } } set_variable = { NSP_newest_mid = nsp_evq } }
	}
}

# 任务 ev_mid 的成功率 +ev_val（百分点，可为负；结果限制在 0..100）
#   已经掷过骰的任务（发射阶段起）照样生效：结局按新成功率重新对照那次的骰子
NSP_ev_mission_rel = {
	if = {
		limit = { NOT = { check_variable = { ev_silent = 1 } } check_variable = { ev_mid > -1 } is_in_array = { array = NSP_missions value = ev_mid } }
		if = { limit = { check_variable = { ev_val < 0 } } custom_effect_tooltip = NSP_EV_TT_MREL_N }
		else = { custom_effect_tooltip = NSP_EV_TT_MREL_P }
	}
	hidden_effect = {
		set_temp_variable = { nsp_mid = ev_mid }
		if = {
			limit = { check_variable = { nsp_mid > -1 } is_in_array = { array = NSP_missions value = nsp_mid } }
			add_to_variable = { NSP_m_rel^nsp_mid = ev_val }
			if = { limit = { check_variable = { NSP_m_rel^nsp_mid < 0 } } set_variable = { NSP_m_rel^nsp_mid = 0 } }
			if = { limit = { check_variable = { NSP_m_rel^nsp_mid > 100 } } set_variable = { NSP_m_rel^nsp_mid = 100 } }
			# 已预掷：让失败类型和新成功率一致（发射动画、发射判定读 fkind）
			if = {
				limit = { check_variable = { NSP_m_roll^num > nsp_mid } check_variable = { NSP_m_roll^nsp_mid > -1 } check_variable = { NSP_m_fkind^num > nsp_mid } }
				if = {
					limit = { check_variable = { NSP_m_roll^nsp_mid < NSP_m_rel^nsp_mid } }
					set_variable = { NSP_m_fkind^nsp_mid = 0 }
				}
				else_if = {
					limit = { check_variable = { NSP_m_fkind^nsp_mid = 0 } }
					set_variable = { NSP_m_fkind^nsp_mid = 5 }   # 已过发射：算在轨失败
					if = {
						limit = { check_variable = { NSP_m_phase^nsp_mid < 6 } }   # 还没发射：和预掷一样随机台上 / 空中 / 在轨
						set_temp_variable_to_random = { var = nsp_k min = 0 max = 100 }
						if = { limit = { check_variable = { nsp_k < 25 } } set_variable = { NSP_m_fkind^nsp_mid = 3 } }
						else_if = { limit = { check_variable = { nsp_k < 60 } } set_variable = { NSP_m_fkind^nsp_mid = 4 } }
					}
				}
			}
		}
	}
	NSP_ev_finish = yes
}

# 任务 ev_mid 的当前阶段推迟 ev_val 天（负数 = 提前；当前阶段最少剩 1 天）
NSP_ev_mission_delay = {
	set_temp_variable = { ev_abs = ev_val } multiply_temp_variable = { ev_abs = -1 }   # 负数分支显示用（= -ev_val），不放进条件块
	if = {
		limit = { NOT = { check_variable = { ev_silent = 1 } } check_variable = { ev_mid > -1 } is_in_array = { array = NSP_missions value = ev_mid } }
		if = { limit = { check_variable = { ev_val < 0 } } custom_effect_tooltip = NSP_EV_TT_MDAY_N }
		else = { custom_effect_tooltip = NSP_EV_TT_MDAY_P }
	}
	hidden_effect = {
		set_temp_variable = { nsp_mid = ev_mid }
		if = {
			limit = { check_variable = { nsp_mid > -1 } is_in_array = { array = NSP_missions value = nsp_mid } }
			set_temp_variable = { nsp_old = NSP_m_phase_left^nsp_mid }
			add_to_variable = { NSP_m_phase_left^nsp_mid = ev_val }
			if = { limit = { check_variable = { NSP_m_phase_left^nsp_mid < 1 } } set_variable = { NSP_m_phase_left^nsp_mid = 1 } }
			set_temp_variable = { nsp_dd = NSP_m_phase_left^nsp_mid }
			subtract_from_temp_variable = { nsp_dd = nsp_old }
			add_to_variable = { NSP_m_total_left^nsp_mid = nsp_dd }
			if = { limit = { check_variable = { NSP_m_total_left^nsp_mid < 0 } } set_variable = { NSP_m_total_left^nsp_mid = 0 } }
		}
	}
	NSP_ev_finish = yes
}
