	while_loop_effect = { limit = { check_variable = { NSP_rd_l2lu^num < nsp_ln } } add_to_array = { NSP_rd_l2lu = 1 } }
	while_loop_effect = { limit = { check_variable = { NSP_rd_l2ld^num < nsp_ln } } add_to_array = { NSP_rd_l2ld = 1 } }
	# 小科技出线：先整体清空（父项目 id 可能比子项目大，不能在主循环里按 p 清）
	while_loop_effect = { limit = { check_variable = { NSP_rd_ltail^num < nsp_ln } } add_to_array = { NSP_rd_ltail = 1 } }
	for_loop_effect = { start = 0 end = nsp_ln value = p  set_variable = { NSP_rd_ltail^p = 1 } }
	for_loop_effect = {
		start = 0 end = nsp_ln value = p
		set_variable = { NSP_rd_l1h0^p = 1 }
		set_variable = { NSP_rd_l1h1^p = 1 }
		set_variable = { NSP_rd_l1h2^p = 1 }
		set_variable = { NSP_rd_l1h3^p = 1 }
		set_variable = { NSP_rd_l1h4^p = 1 }
		set_variable = { NSP_rd_l1h5^p = 1 }
		set_variable = { NSP_rd_l1h6^p = 1 }
		set_variable = { NSP_rd_l2h0^p = 1 }
		set_variable = { NSP_rd_l2h1^p = 1 }
		set_variable = { NSP_rd_l2h2^p = 1 }
		set_variable = { NSP_rd_l2h3^p = 1 }
		set_variable = { NSP_rd_l2h4^p = 1 }
		set_variable = { NSP_rd_l2h5^p = 1 }
		set_variable = { NSP_rd_l2h6^p = 1 }
		set_variable = { NSP_rd_l1vu^p = 1 }
		set_variable = { NSP_rd_l1vd^p = 1 }
		set_variable = { NSP_rd_l1lu^p = 1 }
		set_variable = { NSP_rd_l1ld^p = 1 }
		set_variable = { NSP_rd_l2vu^p = 1 }
		set_variable = { NSP_rd_l2vd^p = 1 }
		set_variable = { NSP_rd_l2lu^p = 1 }
		set_variable = { NSP_rd_l2ld^p = 1 }
		set_temp_variable = { nsp_lc = 0 }
		if = { limit = { check_variable = { NSP_rd_state^p > 0 } check_variable = { NSP_rd_state^p < 4 } } set_temp_variable = { nsp_lc = NSP_rd_state^p } }
		# ---- req1 ----
		set_temp_variable = { nsp_lq = NSP_rd_req1^p }
		if = {
			limit = { check_variable = { nsp_lq > -1 } NOT = { check_variable = { nsp_lq = NSP_rd_blank } } }
			set_temp_variable = { nsp_ldx = NSP_rd_col^p }
			subtract_from_temp_variable = { nsp_ldx = NSP_rd_col^nsp_lq }
			set_temp_variable = { nsp_ldy = NSP_rd_row^p }
			subtract_from_temp_variable = { nsp_ldy = NSP_rd_row^nsp_lq }
			if = {
				limit = { check_variable = { nsp_ldx > 0 } check_variable = { nsp_ldx < 5 } check_variable = { nsp_ldy > -4 } check_variable = { nsp_ldy < 4 } }
				# L 形可用？dx = 1、dy ≠ 0，且父列从父行朝子行走（不含父行，含子行）每格都空
				set_temp_variable = { nsp_lL = 0 }
				if = {
					limit = { check_variable = { nsp_ldx = 1 } NOT = { check_variable = { nsp_ldy = 0 } } check_variable = { NSP_rd_small^nsp_lq = 0 } }   # 小科技父卡没有底边可落，走 Z 形
					set_temp_variable = { nsp_lL = 1 }
					set_temp_variable = { nsp_lst = 1 }
					if = { limit = { check_variable = { nsp_ldy < 0 } } set_temp_variable = { nsp_lst = -1 } }
					set_temp_variable = { nsp_lrow = NSP_rd_row^nsp_lq }
					set_temp_variable = { nsp_lrow_end = NSP_rd_row^p }
					set_temp_variable = { nsp_lbase = NSP_rd_trk^p }
					multiply_temp_variable = { nsp_lbase = 40 }
					add_to_temp_variable = { nsp_lbase = NSP_rd_col^nsp_lq }

						limit = { NOT = { check_variable = { nsp_lrow = nsp_lrow_end } } }
						add_to_temp_variable = { nsp_lrow = nsp_lst }
						set_temp_variable = { nsp_lix = nsp_lrow }
						multiply_temp_variable = { nsp_lix = 8 }
						add_to_temp_variable = { nsp_lix = nsp_lbase }
						set_temp_variable = { nsp_lcell = NSP_rd_index^nsp_lix }
						if = { limit = { NOT = { check_variable = { nsp_lcell = NSP_rd_blank } } } set_temp_variable = { nsp_lL = 0 } }
					}
				}
				set_temp_variable = { nsp_lady = nsp_ldy }
				if = { limit = { check_variable = { nsp_lady < 0 } } multiply_temp_variable = { nsp_lady = -1 } }
				set_temp_variable = { nsp_lv = nsp_lc }
				multiply_temp_variable = { nsp_lv = 3 }
				add_to_temp_variable = { nsp_lv = nsp_lady }
				add_to_temp_variable = { nsp_lv = 1 }
				if = {
					limit = { check_variable = { nsp_lL = 1 } }
					if = { limit = { check_variable = { nsp_ldy > 0 } } set_variable = { NSP_rd_l1lu^p = nsp_lv } }
					else = { set_variable = { NSP_rd_l1ld^p = nsp_lv } }
				}
				else = {
					set_temp_variable = { nsp_lh = nsp_lc }
					multiply_temp_variable = { nsp_lh = 4 }
					add_to_temp_variable = { nsp_lh = nsp_ldx }
					add_to_temp_variable = { nsp_lh = 1 }
					# 父项目是小科技：它的出线颜色取所有子项目里状态最高的那条
					if = {
						limit = { check_variable = { NSP_rd_small^nsp_lq = 1 } }
						set_temp_variable = { nsp_lt = nsp_lc }
						add_to_temp_variable = { nsp_lt = 2 }
						set_temp_variable = { nsp_lt0 = NSP_rd_ltail^nsp_lq }
						if = { limit = { check_variable = { nsp_lt > nsp_lt0 } } set_variable = { NSP_rd_ltail^nsp_lq = nsp_lt } }
					}
					if = { limit = { check_variable = { nsp_ldy = -3 } } set_variable = { NSP_rd_l1h0^p = nsp_lh } }
					if = { limit = { check_variable = { nsp_ldy = -2 } } set_variable = { NSP_rd_l1h1^p = nsp_lh } }
					if = { limit = { check_variable = { nsp_ldy = -1 } } set_variable = { NSP_rd_l1h2^p = nsp_lh } }
					if = { limit = { check_variable = { nsp_ldy = 0 } } set_variable = { NSP_rd_l1h3^p = nsp_lh } }
					if = { limit = { check_variable = { nsp_ldy = 1 } } set_variable = { NSP_rd_l1h4^p = nsp_lh } }
					if = { limit = { check_variable = { nsp_ldy = 2 } } set_variable = { NSP_rd_l1h5^p = nsp_lh } }
					if = { limit = { check_variable = { nsp_ldy = 3 } } set_variable = { NSP_rd_l1h6^p = nsp_lh } }
					if = {
						limit = { check_variable = { nsp_ldy > -1 } }
						set_temp_variable = { nsp_lu = nsp_lc }
						multiply_temp_variable = { nsp_lu = 4 }
						add_to_temp_variable = { nsp_lu = nsp_ldy }
						add_to_temp_variable = { nsp_lu = 2 }
						set_variable = { NSP_rd_l1vu^p = nsp_lu }
					}
					else = { set_variable = { NSP_rd_l1vd^p = nsp_lv } }
				}
			}
		}
		# ---- req2 ----
		set_temp_variable = { nsp_lq = NSP_rd_req2^p }
		if = {
			limit = { check_variable = { nsp_lq > -1 } NOT = { check_variable = { nsp_lq = NSP_rd_blank } } }
			set_temp_variable = { nsp_ldx = NSP_rd_col^p }
			subtract_from_temp_variable = { nsp_ldx = NSP_rd_col^nsp_lq }
			set_temp_variable = { nsp_ldy = NSP_rd_row^p }
			subtract_from_temp_variable = { nsp_ldy = NSP_rd_row^nsp_lq }
			if = {
				limit = { check_variable = { nsp_ldx > 0 } check_variable = { nsp_ldx < 5 } check_variable = { nsp_ldy > -4 } check_variable = { nsp_ldy < 4 } }
				# L 形可用？dx = 1、dy ≠ 0，且父列从父行朝子行走（不含父行，含子行）每格都空
				set_temp_variable = { nsp_lL = 0 }
				if = {
					limit = { check_variable = { nsp_ldx = 1 } NOT = { check_variable = { nsp_ldy = 0 } } check_variable = { NSP_rd_small^nsp_lq = 0 } }   # 小科技父卡没有底边可落，走 Z 形
					set_temp_variable = { nsp_lL = 1 }
					set_temp_variable = { nsp_lst = 1 }
					if = { limit = { check_variable = { nsp_ldy < 0 } } set_temp_variable = { nsp_lst = -1 } }
					set_temp_variable = { nsp_lrow = NSP_rd_row^nsp_lq }
					set_temp_variable = { nsp_lrow_end = NSP_rd_row^p }
					set_temp_variable = { nsp_lbase = NSP_rd_trk^p }
					multiply_temp_variable = { nsp_lbase = 40 }
					add_to_temp_variable = { nsp_lbase = NSP_rd_col^nsp_lq }
					while_loop_effect = {
						limit = { NOT = { check_variable = { nsp_lrow = nsp_lrow_end } } }
						add_to_temp_variable = { nsp_lrow = nsp_lst }
						set_temp_variable = { nsp_lix = nsp_lrow }
						multiply_temp_variable = { nsp_lix = 8 }
						add_to_temp_variable = { nsp_lix = nsp_lbase }
						set_temp_variable = { nsp_lcell = NSP_rd_index^nsp_lix }
						if = { limit = { NOT = { check_variable = { nsp_lcell = NSP_rd_blank } } } set_temp_variable = { nsp_lL = 0 } }
					}
				}
				set_temp_variable = { nsp_lady = nsp_ldy }
				if = { limit = { check_variable = { nsp_lady < 0 } } multiply_temp_variable = { nsp_lady = -1 } }
				set_temp_variable = { nsp_lv = nsp_lc }
				multiply_temp_variable = { nsp_lv = 3 }
				add_to_temp_variable = { nsp_lv = nsp_lady }
				add_to_temp_variable = { nsp_lv = 1 }
				if = {
					limit = { check_variable = { nsp_lL = 1 } }
					if = { limit = { check_variable = { nsp_ldy > 0 } } set_variable = { NSP_rd_l2lu^p = nsp_lv } }
					else = { set_variable = { NSP_rd_l2ld^p = nsp_lv } }
				}
				else = {
					set_temp_variable = { nsp_lh = nsp_lc }
					multiply_temp_variable = { nsp_lh = 4 }
					add_to_temp_variable = { nsp_lh = nsp_ldx }
					add_to_temp_variable = { nsp_lh = 1 }
					# 父项目是小科技：它的出线颜色取所有子项目里状态最高的那条
					if = {
						limit = { check_variable = { NSP_rd_small^nsp_lq = 1 } }
						set_temp_variable = { nsp_lt = nsp_lc }
						add_to_temp_variable = { nsp_lt = 2 }
						set_temp_variable = { nsp_lt0 = NSP_rd_ltail^nsp_lq }
						if = { limit = { check_variable = { nsp_lt > nsp_lt0 } } set_variable = { NSP_rd_ltail^nsp_lq = nsp_lt } }
					}
					if = { limit = { check_variable = { nsp_ldy = -3 } } set_variable = { NSP_rd_l2h0^p = nsp_lh } }
					if = { limit = { check_variable = { nsp_ldy = -2 } } set_variable = { NSP_rd_l2h1^p = nsp_lh } }
					if = { limit = { check_variable = { nsp_ldy = -1 } } set_variable = { NSP_rd_l2h2^p = nsp_lh } }
					if = { limit = { check_variable = { nsp_ldy = 0 } } set_variable = { NSP_rd_l2h3^p = nsp_lh } }
					if = { limit = { check_variable = { nsp_ldy = 1 } } set_variable = { NSP_rd_l2h4^p = nsp_lh } }
					if = { limit = { check_variable = { nsp_ldy = 2 } } set_variable = { NSP_rd_l2h5^p = nsp_lh } }

						add_to_temp_variable = { nsp_lu = nsp_ldy }
						add_to_temp_variable = { nsp_lu = 2 }
						set_variable = { NSP_rd_l2vu^p = nsp_lu }
					}
					else = { set_variable = { NSP_rd_l2vd^p = nsp_lv } }
				}
			}
		}
	}

# 切页：同 NSP_open_planner 的写法（flag 改完再刷一次）
NSP_open_rd = {
	NSP_rd_recalc = yes
	clr_country_flag = NSP_rd_pop_open   # 每次进研发页弹窗先收起
	set_country_flag = NSP_rd_open
	add_to_variable = { NSP_dirty = 1 }
}

NSP_close_rd = {
	clr_country_flag = NSP_rd_open
	clr_country_flag = NSP_rd_pop_open
	NSP_recalc_home = yes
	add_to_variable = { NSP_dirty = 1 }
}

NSP_rd_set_track = {
	# 调用前 set_variable = { NSP_rd_track = X }
	NSP_rd_recalc = yes
	NSP_rd_pick_sel = yes
	NSP_rd_recalc = yes
	clr_country_flag = NSP_rd_pop_open   # 切分类收起弹窗（选中项会被 pick_sel 换掉）
	add_to_variable = { NSP_dirty = 1 }
}

# 切分类后挑一个默认选中：优先研发中，其次可研发，否则该分类第一个项目
NSP_rd_pick_sel = {
	set_temp_variable = { nsp_best = -1 }
	set_temp_variable = { nsp_best_score = -1 }
	for_loop_effect = {
		start = 0 end = NSP_rd_count value = p
		if = {
			limit = { check_variable = { NSP_rd_trk^p = NSP_rd_track } }
			set_temp_variable = { nsp_score = 0 }
			if = { limit = { check_variable = { NSP_rd_state^p = 1 } } set_temp_variable = { nsp_score = 2 } }
			if = { limit = { check_variable = { NSP_rd_state^p = 2 } } set_temp_variable = { nsp_score = 3 } }
			if = {
				limit = { check_variable = { nsp_score > nsp_best_score } }
				set_temp_variable = { nsp_best = p }
				set_temp_variable = { nsp_best_score = nsp_score }
			}
		}
	}
	set_variable = { NSP_rd_sel = nsp_best }
}

NSP_rd_start = {
	if = {
		limit = { NSP_rd_can_start = yes }
		set_temp_variable = { p = NSP_rd_sel }
		subtract_from_variable = { NSP_space_budget = NSP_rd_sel_cost }
		set_variable = { NSP_rd_state^p = 2 }
		set_variable = { NSP_rd_prog^p = 0 }
		set_variable = { NSP_rd_deff^p = NSP_rd_sel_days }
		add_to_array = { NSP_rd_queue = p }
		NSP_rd_recalc = yes
		add_to_variable = { NSP_dirty = 1 }
	}
}

NSP_rd_recalc = {
	# ---- 当前时代（TNO 1962 开局 → 1961 时代；可以研发下一个时代的项目，但工期 ×1.5）----
	set_variable = { NSP_rd_era_now = 0 }
	if = { limit = { date > 1961.1.1 } set_variable = { NSP_rd_era_now = 1 } }
	if = { limit = { date > 1964.1.1 } set_variable = { NSP_rd_era_now = 2 } }
	if = { limit = { date > 1967.1.1 } set_variable = { NSP_rd_era_now = 3 } }
	if = { limit = { date > 1970.1.1 } set_variable = { NSP_rd_era_now = 4 } }
	if = { limit = { date > 1973.1.1 } set_variable = { NSP_rd_era_now = 5 } }
	if = { limit = { date > 1976.1.1 } set_variable = { NSP_rd_era_now = 6 } }
	if = { limit = { date > 1979.1.1 } set_variable = { NSP_rd_era_now = 7 } }
	set_temp_variable = { nsp_lim = NSP_rd_era_now }
	add_to_temp_variable = { nsp_lim = 1 }

	# ---- 状态 0/1 由前置 + 时代决定 ----
	for_loop_effect = {
		start = 0 end = NSP_rd_count value = p
		if = {
			limit = { check_variable = { NSP_rd_state^p < 2 } }
			set_temp_variable = { nsp_ok = 1 }
			set_temp_variable = { nsp_r = NSP_rd_req1^p }
			if = { limit = { check_variable = { nsp_r > -1 } } if = { limit = { NOT = { check_variable = { NSP_rd_state^nsp_r = 3 } } } set_temp_variable = { nsp_ok = 0 } } }
			set_temp_variable = { nsp_r = NSP_rd_req2^p }
			if = { limit = { check_variable = { nsp_r > -1 } } if = { limit = { NOT = { check_variable = { NSP_rd_state^nsp_r = 3 } } } set_temp_variable = { nsp_ok = 0 } } }
			if = { limit = { check_variable = { NSP_rd_col^p > nsp_lim } } set_temp_variable = { nsp_ok = 0 } }
			set_variable = { NSP_rd_state^p = nsp_ok }
		}
	}

	# ---- 当前分类的格位视图（COLS×ROWS）----
	clear_array = NSP_rd_view
	set_temp_variable = { nsp_base = NSP_rd_track }
	multiply_temp_variable = { nsp_base = 40 }
	for_loop_effect = {
		start = 0 end = 40 value = k
		set_temp_variable = { nsp_ix = nsp_base }
		add_to_temp_variable = { nsp_ix = k }
		add_to_array = { NSP_rd_view = NSP_rd_index^nsp_ix }
	}

	# ---- 页签 / 连线帧 ----
	set_variable = { NSP_rd_f_tab0 = 1 } if = { limit = { check_variable = { NSP_rd_track = 0 } } set_variable = { NSP_rd_f_tab0 = 2 } }
	set_variable = { NSP_rd_f_tab1 = 1 } if = { limit = { check_variable = { NSP_rd_track = 1 } } set_variable = { NSP_rd_f_tab1 = 2 } }
	set_variable = { NSP_rd_f_tab2 = 1 } if = { limit = { check_variable = { NSP_rd_track = 2 } } set_variable = { NSP_rd_f_tab2 = 2 } }
	set_variable = { NSP_rd_f_tab3 = 1 } if = { limit = { check_variable = { NSP_rd_track = 3 } } set_variable = { NSP_rd_f_tab3 = 2 } }
	set_variable = { NSP_rd_f_tab4 = 1 } if = { limit = { check_variable = { NSP_rd_track = 4 } } set_variable = { NSP_rd_f_tab4 = 2 } }
	set_variable = { NSP_rd_f_tab5 = 1 } if = { limit = { check_variable = { NSP_rd_track = 5 } } set_variable = { NSP_rd_f_tab5 = 2 } }
	set_variable = { NSP_rd_f_links = NSP_rd_track }
	add_to_variable = { NSP_rd_f_links = 1 }

	# ---- 研发槽条（7 格）：研发中 / 空闲 / 未解锁 ----
	set_variable = { NSP_rd_q_n = 0 }
	for_each_loop = { array = NSP_rd_queue value = q  add_to_variable = { NSP_rd_q_n = 1 } }
	# 还没拿到的研发槽来源，按顺序填给未解锁的格子：1004 / 1007 / 1010 = 科技第 4 / 7 / 10 级里程碑，其余 = 带研发槽奖励的研发 id
	clear_array = NSP_rd_q_lksrc
	if = { limit = { NOT = { has_country_flag = NSP_tier_slot4 } } add_to_array = { NSP_rd_q_lksrc = 1004 } }
	if = { limit = { NOT = { has_country_flag = NSP_tier_slot7 } } add_to_array = { NSP_rd_q_lksrc = 1007 } }
	if = { limit = { NOT = { has_country_flag = NSP_tier_slot10 } } add_to_array = { NSP_rd_q_lksrc = 1010 } }
	for_each_loop = {
		array = NSP_rd_bonus value = t_b index = t_i
		if = { limit = { check_variable = { t_b = 1 } NOT = { check_variable = { NSP_rd_state^t_i = 3 } } } add_to_array = { NSP_rd_q_lksrc = t_i } }
	}
	# 显示格数 = 研发槽上限 + 还能拿到的，最多 7
	set_variable = { NSP_rd_q_shown = NSP_rd_slots }
	add_to_variable = { NSP_rd_q_shown = NSP_rd_q_lksrc^num }
	clamp_variable = { var = NSP_rd_q_shown min = 0 max = 7 }
	# 第 1 格
	set_variable = { NSP_rd_q_id0 = NSP_rd_blank }
	set_variable = { NSP_rd_q_prog0 = 0 }
	set_variable = { NSP_rd_q_cf0 = 3 }
	set_variable = { NSP_rd_q_lk0 = 0 }
	if = { limit = { check_variable = { NSP_rd_slots > 0 } } set_variable = { NSP_rd_q_cf0 = 1 } }
	else = {
		set_temp_variable = { t_j = 0 }
		subtract_from_temp_variable = { t_j = NSP_rd_slots }
		if = { limit = { check_variable = { NSP_rd_q_lksrc^num > t_j } } set_variable = { NSP_rd_q_lk0 = NSP_rd_q_lksrc^t_j } }
	}
	if = {
		limit = { check_variable = { NSP_rd_q_n > 0 } }
		set_variable = { NSP_rd_q_cf0 = 2 }
		set_variable = { NSP_rd_q_id0 = NSP_rd_queue^0 }
		set_temp_variable = { nsp_q = NSP_rd_q_id0 }
		set_variable = { NSP_rd_q_prog0 = NSP_rd_prog^nsp_q }
		round_variable = NSP_rd_q_prog0

#############################################################################
#  太空科技等级系统
#
#  门槛：第 N 级升 N+1 级需要 30 + 10×N（2 级 50 … 9 级 120）。1962 开局 = 第 2 级 0/50。
#        全部研发约 494 点 + 任务经验，正常一局能走到 9～10 级（旧版 +50 递增，打完全部研发也只到 5 级）。
#  任务经验（任务成功时）：近地 5 · 极地 6 · 交会 / 地球静止 8 · 逃逸 10 · 地月转移 12；载人 +3。
#  被动加成（按等级现算，幂等；存在 NSP_tier_rdspd / NSP_tier_rel）：
#        研发速度 +4% / 级（第 1 级起算）    → NSP_rd_recalc 里换算成研发工期
#        火箭可靠性 +1 / 级（第 1 级起算）    → NSP_recalc_planner
#  里程碑（一次性，flag 记账）：第 4 / 7 / 10 级各 +1 研发槽。
#  升级奖励：拨款 300 + 50×新等级（百万美元），支持度 +3。
#############################################################################

# 门槛 = 30 + 10 × 等级
NSP_tier_set_max = {
	set_variable = { NSP_tech_max = NSP_tech_tier }
	multiply_variable = { NSP_tech_max = 10 }
	add_to_variable = { NSP_tech_max = 30 }
}

# 被动加成 + 里程碑研发槽（幂等，每日 / 升级 / 开局调用）
NSP_tier_refresh = {
	set_variable = { NSP_tier_rdspd = NSP_tech_tier }
	subtract_from_variable = { NSP_tier_rdspd = 1 }
	if = { limit = { check_variable = { NSP_tier_rdspd < 0 } } set_variable = { NSP_tier_rdspd = 0 } }
	set_variable = { NSP_tier_rel = NSP_tier_rdspd }
	multiply_variable = { NSP_tier_rdspd = 4 }
	if = { limit = { check_variable = { NSP_tech_tier > 3 } NOT = { has_country_flag = NSP_tier_slot4 } }  set_country_flag = NSP_tier_slot4  add_to_variable = { NSP_rd_slots = 1 } }
	if = { limit = { check_variable = { NSP_tech_tier > 6 } NOT = { has_country_flag = NSP_tier_slot7 } }  set_country_flag = NSP_tier_slot7  add_to_variable = { NSP_rd_slots = 1 } }
	if = { limit = { check_variable = { NSP_tech_tier > 9 } NOT = { has_country_flag = NSP_tier_slot10 } } set_country_flag = NSP_tier_slot10 add_to_variable = { NSP_rd_slots = 1 } }
	# 显示用：下一级的奖励
	set_variable = { NSP_tier_next_pay = NSP_tech_tier }
	add_to_variable = { NSP_tier_next_pay = 1 }
	multiply_variable = { NSP_tier_next_pay = 50 }
	add_to_variable = { NSP_tier_next_pay = 300 }
	set_variable = { NSP_tier_next_payb = NSP_tier_next_pay }
	divide_variable = { NSP_tier_next_payb = 1000 }
	# 等级圆圈 tooltip 用：下一级累计加成、还差多少、已得研发槽
	set_variable = { NSP_tier_next_rdspd = NSP_tier_rdspd } add_to_variable = { NSP_tier_next_rdspd = 4 }
	set_variable = { NSP_tier_next_rel = NSP_tier_rel } add_to_variable = { NSP_tier_next_rel = 1 }
	set_variable = { NSP_tier_need = NSP_tech_max } subtract_from_variable = { NSP_tier_need = NSP_tech }
	if = { limit = { check_variable = { NSP_tier_need < 0 } } set_variable = { NSP_tier_need = 0 } }
	set_variable = { NSP_tier_slots_got = 0 }
	if = { limit = { has_country_flag = NSP_tier_slot4 } add_to_variable = { NSP_tier_slots_got = 1 } }
	if = { limit = { has_country_flag = NSP_tier_slot7 } add_to_variable = { NSP_tier_slots_got = 1 } }
	if = { limit = { has_country_flag = NSP_tier_slot10 } add_to_variable = { NSP_tier_slots_got = 1 } }
}

# 升级（可连升；只弹一次 nsp.11）
NSP_tier_levelup = {
	set_temp_variable = { nsp_up = 0 }
	while_loop_effect = {
		limit = { NOT = { check_variable = { NSP_tech < NSP_tech_max } } }
		subtract_from_variable = { NSP_tech = NSP_tech_max }
		add_to_variable = { NSP_tech_tier = 1 }
		NSP_tier_set_max = yes
		set_variable = { NSP_tier_last_pay = NSP_tech_tier }
		multiply_variable = { NSP_tier_last_pay = 50 }
		add_to_variable = { NSP_tier_last_pay = 300 }
		add_to_variable = { NSP_space_budget = NSP_tier_last_pay }
		add_to_variable = { NSP_popularity = 3 }
		set_temp_variable = { nsp_up = 1 }
	}
	if = { limit = { check_variable = { NSP_popularity > 100 } } set_variable = { NSP_popularity = 100 } }
	NSP_tier_refresh = yes
	if = {
		limit = { check_variable = { nsp_up = 1 } }
		set_variable = { NSP_tier_last_payb = NSP_tier_last_pay }
		divide_variable = { NSP_tier_last_payb = 1000 }
		country_event = { id = nsp.11 }
	}
}

# 任务成功经验（nsp.1 调用；读 NSP_last_mission）
NSP_tier_mission_xp = {
	set_temp_variable = { nsp_lm = NSP_last_mission }
	set_temp_variable = { nsp_xp = 5 }
	if = {
		limit = { check_variable = { NSP_m_dest^num > nsp_lm } }
		set_temp_variable = { nsp_d = NSP_m_dest^nsp_lm }
		if = { limit = { check_variable = { NSP_dest_xp^num > nsp_d } } set_temp_variable = { nsp_xp = NSP_dest_xp^nsp_d } }   # 经验改查 NSP_dest_xp
	}
	if = {
		limit = { check_variable = { NSP_m_payload^num > nsp_lm } }
		set_temp_variable = { nsp_p = NSP_m_payload^nsp_lm }
		if = { limit = { check_variable = { NSP_pl_crewed^nsp_p = 1 } } add_to_temp_variable = { nsp_xp = 3 } }
	}
	add_to_variable = { NSP_tech = nsp_xp }
	set_variable = { NSP_tier_last_xp = nsp_xp }
	NSP_tier_levelup = yes
}
