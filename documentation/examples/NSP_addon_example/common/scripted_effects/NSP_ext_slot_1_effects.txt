# 示例：先驱者 5 号用雷神-艾布尔发射时可靠性 +5（演示怎么改规划器结果）
NSP_slot1_on_recalc = {
	if = {
		limit = { check_variable = { p = 50 } check_variable = { ve = 50 } }
		add_to_variable = { NSP_calc_rel_final = 5 }
		if = { limit = { check_variable = { NSP_calc_rel_final > 100 } } set_variable = { NSP_calc_rel_final = 100 } }
	}
}

# 示例：先驱者 5 号成功进入日心轨道 → 额外民意 +3
NSP_slot1_on_success = {
	if = {
		limit = { check_variable = { p = 50 } check_variable = { d = 16 } }
		add_to_variable = { NSP_popularity = 3 }
		if = { limit = { check_variable = { NSP_popularity > 100 } } set_variable = { NSP_popularity = 100 } }
	}
}
